<script>

// Mock data initialization for 28 students

let students = JSON.parse(localStorage.getItem('sr_students')) || Array.from({length: 28}, (_, i) => ({

id: i + 1,

name: `Student ${i + 1}`,

logs: [],

adkarScore: Math.floor(Math.random() * 40) + 60,

medicalNotes: i % 5 === 0 ? "Mild allergy to peanuts" : "None",

achievements: ["Active Participant"]

}));

let activeStudentId = null;

function handleLogin(e) {

e.preventDefault();

const pass = document.getElementById('passcode').value;

if (pass === "1234" || pass.length > 0) { // simple passcode check

document.getElementById('auth-screen').classList.add('hidden');

document.getElementById('app-container').classList.remove('hidden');

initApp();

} else {

alert('Invalid passcode');

}

}

function logout() {

document.getElementById('auth-screen').classList.remove('hidden');

document.getElementById('app-container').classList.add('hidden');

}

function switchTab(tabName) {

['dashboard', 'students', 'analytics'].forEach(t => {

document.getElementById(`tab-${t}`).classList.add('hidden');

});

document.querySelectorAll('.tab-btn').forEach(btn => {

btn.classList.remove('border-indigo-600', 'text-indigo-600');

btn.classList.add('border-transparent', 'text-slate-600');

});

document.getElementById(`tab-${tabName}`).classList.remove('hidden');

event.target.classList.add('border-indigo-600', 'text-indigo-600');

event.target.classList.remove('border-transparent', 'text-slate-600');

}

function initApp() {

updateStats();

populateStudentDropdown();

renderStudentsGrid();

renderLeaderboard();

}

function updateStats() {

let totalLogs = students.reduce((acc, s) => acc + s.logs.length, 0);

let avgAdkar = Math.round(students.reduce((acc, s) => acc + s.adkarScore, 0) / students.length);

let medicalCount = students.filter(s => s.medicalNotes !== "None").length;

document.getElementById('stat-logs').innerText = totalLogs;

document.getElementById('stat-adkar').innerText = avgAdkar + '%';

document.getElementById('stat-medical').innerText = medicalCount;

localStorage.setItem('sr_students', JSON.stringify(students));

}

function populateStudentDropdown() {

const select = document.getElementById('quick-student-select');

select.innerHTML = students.map(s => `<option value="${s.id}">${s.name}</option>`).join('');

}

function handleQuickLog(e) {

e.preventDefault();

const studentId = parseInt(document.getElementById('quick-student-select').value);

const activity = document.getElementById('quick-activity').value;

const score = document.getElementById('quick-score').value;

const student = students.find(s => s.id === studentId);

if (student) {

student.logs.push({ date: new Date().toLocaleDateString(), activity, score });

updateStats();

document.getElementById('quick-score').value = '';

alert('Record saved successfully!');

renderStudentsGrid();

renderLeaderboard();

}

}

function renderStudentsGrid(filter = "") {

const grid = document.getElementById('student-grid');

const filtered = students.filter(s => s.name.toLowerCase().includes(filter.toLowerCase()));

grid.innerHTML = filtered.map(s => `

<div onclick="openStudentModal(${s.id})" class="bg-white p-4 rounded-xl border border-slate-200 shadow-sm hover:border-indigo-500 cursor-pointer transition">

<h4 class="font-bold text-slate-900">${s.name}</h4>

<p class="text-xs text-slate-500 mt-1">Adkar: ${s.adkarScore}%</p>

<p class="text-xs text-slate-500">Logs: ${s.logs.length}</p>

${s.medicalNotes !== "None" ? '<span class="inline-block mt-2 bg-amber-50 text-amber-700 text-[10px] px-2 py-0.5 rounded font-medium">Medical Flag</span>' : ''}

</div>

`).join('');

}

function filterStudents() {

const query = document.getElementById('search-student').value;

renderStudentsGrid(query);

}

function openStudentModal(id) {

activeStudentId = id;

const student = students.find(s => s.id === id);

document.getElementById('modal-student-name').innerText = student.name;

let html = `

<div class="space-y-3">

<div class="bg-slate-50 p-3 rounded-lg text-xs space-y-1">

<p><strong>Adkar Score:</strong> ${student.adkarScore}%</p>

<p><strong>Medical Notes:</strong> ${student.medicalNotes}</p>

</div>

<h4 class="font-bold text-sm text-slate-800">Activity & Practice Logs</h4>

`;

if (student.logs.length === 0) {

html += `<p class="text-xs text-slate-400">No logs recorded yet.</p>`;

} else {

html += `<ul class="space-y-2">` + student.logs.map(l => `

<li class="bg-white border border-slate-100 p-2 rounded text-xs flex justify-between">

<span><strong>${l.activity}:</strong> ${l.score}</span>

<span class="text-slate-400">${l.date}</span>

</li>

`).join('') + `</ul>`;

}

html += `</div>`;

document.getElementById('modal-body').innerHTML = html;

document.getElementById('student-modal').classList.remove('hidden');

}

function closeModal() {

document.getElementById('student-modal').classList.add('hidden');

activeStudentId = null;

}

function renderLeaderboard() {

const list = document.getElementById('leaderboard-list');

const sorted = [...students].sort((a, b) => b.adkarScore - a.adkarScore).slice(0, 5);

list.innerHTML = sorted.map((s, idx) => `

<div class="flex justify-between items-center text-sm p-2 bg-slate-50 rounded-lg">

<span class="font-medium text-slate-700">#${idx + 1} ${s.name}</span>

<span class="text-indigo-600 font-bold">${s.adkarScore}%</span>

</div>

`).join('');

}

function exportData() {

const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(students));

const dlAnchorElem = document.createElement('a');

dlAnchorElem.setAttribute("href", dataStr);

dlAnchorElem.setAttribute("download", "student_records_backup.json");

dlAnchorElem.click();

}

function exportMasterPDF() {

const { jsPDF } = window.jspdf;

const doc = new jsPDF();

doc.text("Master Student Records Report", 14, 20);

let y = 30;

students.forEach((s, idx) => {

doc.text(`${idx + 1}. ${s.name} - Adkar: ${s.adkarScore}% - Logs: ${s.logs.length}`, 14, y);

y += 10;

if (y > 280) { doc.addPage(); y = 20; }

});

doc.save("master_student_roster.pdf");

}

function exportIndividualPDF() {

if (!activeStudentId) return;

const student = students.find(s => s.id === activeStudentId);

const { jsPDF } = window.jspdf;

const doc = new jsPDF();

doc.text(`Student Report: ${student.name}`, 14, 20);

doc.text(`Adkar Score: ${student.adkarScore}%`, 14, 30);

doc.text(`Medical Notes: ${student.medicalNotes}`, 14, 40);

doc.text("Logs:", 14, 50);

let y = 60;

student.logs.forEach(l => {

doc.text(`- ${l.date} | ${l.activity}: ${l.score}`, 18, y);

y += 10;

});

doc.save(`${student.name}_report.pdf`);

}
